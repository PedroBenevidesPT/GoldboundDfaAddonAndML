# Runtime dependencies for the pipeline, the models and the database layer.
#
# This is the set the worker image installs (docker/app/Dockerfile).  It is
# deliberately narrower than what a developer machine has:
#
#   * no pytest -- that is test-only, see requirements-dev.txt;
#   * no playwright -- that is the wowhead scraper, see requirements-scrape.txt.
#     The scraper is preserved, it just does not belong in a batch worker
#     image that would then carry a browser for no reason.
#
# Versions are the ones this project has actually been run against, not
# guesses.  Each has a floor (the tested version) and a ceiling on the next
# minor, because a container that resolves a newer pandas than the one the
# suite passed on is not reproducible -- the first build of this image picked
# up pandas 3.1.0 while the tested version was 3.0.6.

# --- HTTP (Blizzard API, TSM CSV, webhooks) ---
requests>=2.34

# --- Database -------------------------------------------------------------
# SQLAlchemy is the ORM/migration layer (db/models.py, db/migrations).
# psycopg2 is the driver Postgres needs; it is also what the SQLite-to-Postgres
# compatibility layer talks to when DATABASE_URL is set.
SQLAlchemy>=2.1
alembic>=1.13
psycopg2-binary>=2.9.9
redis>=5.0

# --- Numerics and learning ------------------------------------------------
numpy>=2.5,<2.6
pandas>=3.0.6,<3.2
scikit-learn>=1.9,<2.0
catboost>=1.2,<1.3
joblib>=1.6,<2.0
scipy>=1.18,<1.19
statsmodels>=0.15,<0.16
# lupa is a *runtime* dependency, not a test one: the journal the model
# trains on is a Lua file, and `ml/journalator.py` parses it with a Lua
# runtime.  It was listed as test-only, so the worker image could not read
# the journal and the export died at "the shipped model could not be
# fitted" -- with the file sitting in a mounted volume.
lupa>=2.8,<3.0

# --- Config ---------------------------------------------------------------
python-dotenv>=1.2

# --- Telemetry (optional at runtime, imported lazily) ---------------------
posthog
opentelemetry-api
opentelemetry-sdk
opentelemetry-exporter-otlp
